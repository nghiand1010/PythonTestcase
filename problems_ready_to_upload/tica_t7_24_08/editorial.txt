def count_tiles(a, b):
    if a <= 2 or b <= 2:
        # Nếu a hoặc b <= 2, toàn bộ viên gạch đều là màu xanh
        blue_tiles = a * b
        yellow_tiles = 0
    else:
        # Tổng số viên gạch
        total_tiles = a * b
        # Số viên gạch phần lõi
        core_tiles = (a - 2) * (b - 2)
        # Số viên gạch màu xanh
        blue_tiles = total_tiles - core_tiles
        # Số viên gạch màu vàng
        yellow_tiles = core_tiles
    
    return blue_tiles, yellow_tiles

# Đọc đầu vào
a = int(input())
b = int(input())

# Tính và in ra kết quả
blue, yellow = count_tiles(a, b)
print(f"{blue} {yellow}")
