```python
def sd(n):
    return sum(map(int, str(n)))

def smith(n):
    s1 = sd(n)
    s2 = 0
    m = n
    cnt = 0
    p = 2
    while p * p <= m:
        while m % p == 0:
            s2 += sd(p)
            m //= p
            cnt += 1
        p += 1
    if m > 1:
        s2 += sd(m)
        cnt += 1
    return cnt > 1 and s1 == s2

t = int(input())
for _ in range(t):
    print(1 if smith(int(input())) else 0)
```
