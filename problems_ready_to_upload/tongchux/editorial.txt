```
m = int(input())
n = int(input())
x = int(input())
y = int(input())

g = 2 * x + 3 * y

# Đường chéo chính "\"
a = min(x - 1, y - 1)
b = min(m - x, n - y)

k1 = a + b + 1

dau1 = g - 5 * a
cuoi1 = g + 5 * b

tong1 = k1 * (dau1 + cuoi1) // 2

# Đường chéo phụ "/"
c = min(x - 1, n - y)
d = min(m - x, y - 1)

k2 = c + d + 1

dau2 = g + c
cuoi2 = g - d

tong2 = k2 * (dau2 + cuoi2) // 2

# Trừ ô (x, y) vì bị tính 2 lần
ket_qua = tong1 + tong2 - g

print(ket_qua)
```
