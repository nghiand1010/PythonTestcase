def square_digit_sum(n):
    """Returns the sum of the squares of the digits of n."""
    return sum(int(digit) ** 2 for digit in str(n))

def ends_in_89(n):
    """Returns True if the sequence starting from n ends in 89."""
    while n != 1 and n != 89:
        n = square_digit_sum(n)
    return n == 89

def count_p_ending_in_89(N):
    """Counts how many p < N have sequences ending in 89."""
    count = 0
    for p in range(1, N):
        if ends_in_89(p):
            count += 1
    return count

# Read input
N = int(input().strip())

# Output the result
print(count_p_ending_in_89(N))
