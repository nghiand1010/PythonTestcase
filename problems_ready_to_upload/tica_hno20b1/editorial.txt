def find_min_M(L, R):
    def calculate_sum(a, b):
        """Tính tổng các số từ a đến b"""
        return (b - a + 1) * (a + b) // 2

    total_sum = calculate_sum(L, R)
    left, right = L, R - 1
    best_M = L
    min_diff = float('inf')

    # Tìm kiếm nhị phân
    while left <= right:
        M = (left + right) // 2
        left_sum = calculate_sum(L, M)
        right_sum = total_sum - left_sum
        diff = abs(left_sum - right_sum)

        # Cập nhật kết quả nếu tìm được giá trị tốt hơn
        if diff < min_diff:
            min_diff = diff
            best_M = M

        # Điều chỉnh tìm kiếm nhị phân
        if left_sum < right_sum:
            left = M + 1
        else:
            right = M - 1

    return best_M

# Đọc input và ghi output từ bàn phím
def main():
    L, R = map(int, input().split())
    print(find_min_M(L, R))

if __name__ == "__main__":
    main()
