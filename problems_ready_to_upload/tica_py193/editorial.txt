```python
t=int(input())
for _ in range(t):
    n=int(input()); a=list(map(int,input().split())); b=list(map(int,input().split())); d=[b[i]-a[i] for i in range(n)]; ok=True; i=0
    while i<n and d[i]==0: i+=1
    if i<n:
        if d[i]<=0: ok=False
        val=d[i]
        while i<n and d[i]==val: i+=1
        while i<n:
            if d[i]!=0: ok=False
            i+=1
    print('YES' if ok else 'NO')
```
