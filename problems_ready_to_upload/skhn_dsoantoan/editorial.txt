```
import math

n = int(input())
k = int(input())

# Lưu ý:
# Nếu K = 1 thì mọi số đều chia hết cho 1,
# nên theo đúng đề bài, dãy số an toàn sẽ rỗng.
# Trường hợp này đề bài cần làm rõ.
if k == 1:
    print(0)
else:
    L = k * 10 // math.gcd(k, 10)

    a = []

    # Tìm các số an toàn trong một chu kỳ
    for i in range(1, L + 1):
        if i % k != 0 and i % 10 != k:
            a.append(i)

    m = len(a)

    q = n // m
    r = n % m

    tong_chu_ky = sum(a)

    # Tổng của q chu kỳ đầy đủ
    tong = q * tong_chu_ky

    # Mỗi chu kỳ sau tăng thêm L cho mỗi số an toàn
    tong += m * L * q * (q - 1) // 2

    # Cộng thêm r số đầu tiên của chu kỳ tiếp theo
    tong += q * L * r + sum(a[:r])

    print(tong)
```
