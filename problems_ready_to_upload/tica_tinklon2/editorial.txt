def find_largest_less_than_k(a, k):
    left, right = 0, len(a) - 1
    result = -1  # Biến lưu kết quả

    while left <= right:
        mid = (left + right) // 2
        if a[mid] < k:
            result = a[mid]  # Cập nhật kết quả
            left = mid + 1   # Tìm ở nửa bên phải
        else:
            right = mid - 1  # Tìm ở nửa bên trái

    return result

# Đọc dữ liệu đầu vào
n = int(input())  # Số phần tử của dãy
a = list(map(int, input().split()))  # Dãy a đã sắp xếp tăng dần
k = int(input())  # Số K

# Tìm kết quả
result = find_largest_less_than_k(a, k)

# In kết quả
if result == -1:
    print("KHONG CO")
else:
    print(result)
