import sys
sys.stdin = open("LUCKY.INP","r")
sys.stdout = open("LUCKY.OUT","w")
input = sys.stdin.readline

n = int(input())
cnt = [0] * 10

for _ in range(n):
    m = int(input())
    if m == 0:
        root = 0
    else:
        root = 1 + (m - 1) % 9
    cnt[root] += 1

max_freq = -1
ans = 0
for i in range(10):
    if cnt[i] > max_freq:
        max_freq = cnt[i]
        ans = i

print(ans)
