def count_digits(s):
    cnt = [0] * 10
    for ch in s:
        cnt[ord(ch) - 48] += 1
    return cnt

with open("STICKERS.INP", "r", encoding="utf-8") as f:
    T = f.readline().strip()
    S = f.readline().strip()

cntT = count_digits(T)
cntS = count_digits(S)

INF = 10**18
ans = INF

for d in [0, 1, 3, 4, 7, 8]:
    if cntS[d] > 0:
        ans = min(ans, cntT[d] // cntS[d])

need25 = cntS[2] + cntS[5]
if need25 > 0:
    ans = min(ans, (cntT[2] + cntT[5]) // need25)

need69 = cntS[6] + cntS[9]
if need69 > 0:
    ans = min(ans, (cntT[6] + cntT[9]) // need69)

if ans == INF:
    ans = 0

with open("STICKERS.OUT", "w", encoding="utf-8") as f:
    f.write(str(ans))
