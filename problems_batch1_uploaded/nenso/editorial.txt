# NÉN SỐ - đọc ghi file: nenso.inp -> nenso.out

with open("nenso.inp", "r") as f:
    s = f.read().strip()

# đề cho 0 < N <= 10^9, nhưng ta vẫn xử lý cả N = 0 cho chắc
N = int(s)

if N == 0:
    ans = 0
else:
    ans = 1 + (N - 1) % 9   # công thức digital root

with open("nenso.out", "w") as f:
    f.write(str(ans))
