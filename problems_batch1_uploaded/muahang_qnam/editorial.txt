import sys
sys.stdin = open("BUY.INP","r")
sys.stdout = open("BUY.OUT","w")
n = int(sys.stdin.readline().strip())
a = list(map(int, sys.stdin.readline().strip().split()))
a.sort()
cur = 0
ans = 0
for x in a:
    if x > cur + 1:
        break
    cur += x
ans = cur + 1
sys.stdout.write(str(ans))
