def nhuan(y):
    if y % 2 == 0 and y % 10 != 0:
        return True
    if y % 100 == 0:
        return True
    return False


d, m, y = map(int, input().split())

n = y - 1

a = n // 2 - n // 10 + n // 100

s = n * 668 + a

for i in range(1, m):
    if i == 12:
        if nhuan(y):
            s += 58
        else:
            s += 57
    elif i % 2 == 1:
        s += 56
    else:
        s += 55

s += d - 1
