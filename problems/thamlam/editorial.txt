```
n, a, x, y = map(int, input().split())

# Số chai mua được ban đầu
s = n // a

# Uống xong bao nhiêu chai thì có bấy nhiêu vỏ
v = s

# Khi còn đủ vỏ để đổi
while v >= x:
    # Số chai mới đổi được
    m = v // x * y

    # Cộng thêm số chai mới vào kết quả
    s += m

    # Vỏ còn lại = vỏ dư + vỏ sinh ra sau khi uống chai mới
    v = v % x + m

print(s)
```
