def tong_uoc(x):
    s = 0
    i = 1
    while i * i <= x:
        if x % i == 0:
            s += i
            if i * i != x:
                s += x // i
        i += 1
    return s

t = int(input())
for _ in range(t):
    n = int(input())
    ans = 0
    i = 1
    while i * i <= n:
        if n % i == 0:
            ans += tong_uoc(i)
            if i * i != n:
                ans += tong_uoc(n // i)
        i += 1
    print(ans)
