```python
import math

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    g = 0
    for x in a:
        g = math.gcd(g, x)
    ans = 0
    i = 1
    while i * i <= g:
        if g % i == 0:
            ans += 1
            if i != g // i:
                ans += 1
        i += 1
    print(ans)
```
