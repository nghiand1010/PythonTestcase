def is_leap_year(year):
    if (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0):
        return True
    return False

def get_days_in_month(month, year):
    if month in [1, 3, 5, 7, 8, 10, 12]:
        return 31
    elif month in [4, 6, 9, 11]:
        return 30
    elif month == 2:
        return 29 if is_leap_year(year) else 28
    else:
        return 0

def is_valid_date(day, month, year):
    if year < 2000 or year > 2023:
        return False
    if month < 1 or month > 12:
        return False
    max_day = get_days_in_month(month, year)
    if day < 1 or day > max_day:
        return False
    
    # Kiểm tra phạm vi cụ thể đến 04/06/2023
    if year == 2023:
        if month > 6:
            return False
        if month == 6 and day > 4:
            return False
    return True

def find_next_date(input_str):
    if len(input_str) != 8 or not input_str.isdigit():
        return "Khong tim duoc"
    
    day = int(input_str[:2])
    month = int(input_str[2:4])
    year = int(input_str[4:8])
    
    # Kiểm tra ngày nhập vào có hợp lệ không
    if not is_valid_date(day, month, year):
        return "Khong tim duoc"
    
    # Tính ngày tiếp theo
    max_day = get_days_in_month(month, year)
    day += 1
    if day > max_day:
        day = 1
        month += 1
        if month > 12:
            month = 1
            year += 1
    
    # Kiểm tra ngày tiếp theo có hợp lệ không
    if not is_valid_date(day, month, year):
        return "Khong tim duoc"
    
    # Định dạng kết quả
    next_day = f"{day:02d}{month:02d}{year:04d}"
    return next_day

# Nhập input và xuất output
input_date = input().strip()
print(find_next_date(input_date))
