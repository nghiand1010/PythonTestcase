from math import gcd

def lcm(x, y):
    """Tìm bội chung nhỏ nhất (LCM) của hai số"""
    return x * y // gcd(x, y)

def count_valid_numbers(n, a, b, c):
    """Tính số lượng các số thỏa mãn điều kiện"""
    # Số lượng chia hết cho từng cặp
    count_ab = n // lcm(a, b)
    count_bc = n // lcm(b, c)
    count_ca = n // lcm(c, a)
    
    # Số lượng chia hết cho cả ba
    count_abc = n // lcm(a, lcm(b, c))
    
    #print(count_ab, count_bc, count_ca, count_abc)
    
    # Số lượng thỏa mãn điều kiện
    result = count_ab + count_bc + count_ca - 3 * count_abc
    return result

# Nhập dữ liệu từ bàn phím
n, a, b, c = map(int, input().split())

# Tính toán và in kết quả
result = count_valid_numbers(n, a, b, c)
print(result)
