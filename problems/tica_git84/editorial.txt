# This code is a simple C++ program that counts how many rows in a 2D array have a sum greater than 1.

n = int(input())
array = []

for i in range(n):
    row = list(map(int, input().split()))
    array.append(row)

sum_greater_than_one_count = 0

for i in range(n):
    row_sum = sum(array[i])
    if row_sum > 1:
        sum_greater_than_one_count += 1

print(sum_greater_than_one_count)
