from collections import deque

def kth_lucky_number(k):
    """
    Trả về số may mắn thứ k (theo thứ tự tăng dần) 
    bằng cách dùng hàng đợi (queue) sinh dần các số may mắn.
    """
    queue = deque()
    # Bắt đầu với hai số 1 chữ số
    queue.append("4")
    queue.append("7")
    
    count = 0
    while queue:
        # Lấy phần tử đầu tiên
        front = queue.popleft()
        count += 1
        
        # Nếu đây là lần pop thứ k, ta trả về front
        if count == k:
            return front
        
        # Thêm hai số mới (lớn hơn) vào queue
        queue.append(front + "4")
        queue.append(front + "7")
    
    # Lý thuyết: ta sẽ luôn tìm được kết quả trước khi queue rỗng,
    # vì số may mắn là vô hạn. Dòng này chỉ để an toàn.
    return None

def solve():
    import sys
    data = sys.stdin.read().strip()
    k = int(data)
    
    # Tìm và in ra số may mắn thứ k
    print(kth_lucky_number(k))
    
solve()
