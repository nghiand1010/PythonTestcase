```python
t = int(input())
for _ in range(t):
    n, x = map(int, input().split())
    max_d = 0
    max_giam = -10 ** 18
    for _ in range(n):
        d, h = map(int, input().split())
        max_d = max(max_d, d)
        max_giam = max(max_giam, d - h)
    if max_d >= x:
        print(1)
    elif max_giam <= 0:
        print(-1)
    else:
        con = x - max_d
        print(1 + (con + max_giam - 1) // max_giam)
```
